# Islanders should prefer islands, Continental should prefer Land, unless theyve taken to the seas
islander_continental_preferences_ai_score_value = {
	value = 1
	if = {
		limit = {
			scope:attacker.culture = {
				OR = {
					has_cultural_tradition = tradition_islander
					has_cultural_tradition = tradition_settlers_of_krehejad
					has_cultural_tradition = tradition_guillate_legacy
				}
			}
		}
		every_target_title = {
			title_province = {
				if = {
					limit = {
						geographical_region = GH_geographic_isles
					}
					add = 1000
				}
				else = { add = -1000 }
			}
		}
	}
	if = {
		limit = {
			OR = {
				scope:attacker = { highest_held_title_tier < tier_duchy } # Counties should be less interested in expanding to the isles
				scope:attacker.culture = {
					NOR = {
						has_cultural_tradition = tradition_islander
						has_cultural_tradition = tradition_settlers_of_krehejad
						has_cultural_tradition = tradition_guillate_legacy
					}
				}
				scope:attacker.house ?= { NOT = { has_house_modifier = isles_naval_expedition_modifier} }
			}
		}
		every_target_title = {
			title_province = {
				if = {
					limit = {
						geographical_region = GH_geographic_isles
					}
					add = -100
				}
			}
		}
	}
}

kashirya_conquests_ai_score_value = {
	value = 1
	# Kashiryan is normally surrounded by steppes & tribes, leaving it open for easy outward conquest
	# This keeps them interested on Kashirya proper until developing an empire title
	if = {
		limit = {
			scope:attacker = {
				# Nobody cares about counts :(
				highest_held_title_tier >= tier_duchy
				# And are roughly of Kashiryan heritage
				culture = { has_cultural_tradition = tradition_otachedian_law } # more cultures  have this than a shared tradition
			}
		}
		every_target_title = {
			title_province = {
				if = {
					limit = { geographical_region = GH_geographic_kashirya }
					add = 100
				}
				else_if = {
					limit = { geographical_region = GH_geographic_pirate_coast }
					add = 25
				}
				else_if = {
					limit = { geographical_region = GH_geographic_kasiyrat }
					add = 10
				}
				else_if = {
					limit = { geographical_region = GH_geographic_dahangon }
					add = 5
				}
				else = { add = -1000 }
			}
		}
	}
}

			limit = { scope:actor = { has_legitimacy_flag = extra_increased_alliance_acceptance } }
			add = 50
		}
	}

	# INFLUENCE
	if = {
		limit = { 
			exists = scope:influence
			always = scope:influence 
		}
		add = 10
		multiply = scope:actor.influence_level
		min = 5
	}
	
}

military_strength_alliance_value = {
	value = 1
	subtract = {
		value = scope:recipient.max_military_strength # For foreign diplomacy, consider recipients max
		divide = { value = scope:actor.current_military_strength min = 1 }
	}
	multiply = 20
	min = -1000
	max = 100
}

fp3_struggle_resist_allied_wars_modifier_value = {
	value = 0
	if = {
		limit = {
			OR = {
				AND = {
					scope:actor = { has_trait = fp3_struggle_detractor }
					scope:recipient = { has_trait = fp3_struggle_supporter }
				}
				AND = {
					scope:actor = { has_trait = fp3_struggle_supporter }
					scope:recipient = { has_trait = fp3_struggle_detractor }
				}
			}
		}
		add = -100
	}		
}

evaluate_action_increasing_house_unity_value = {
	value = 0
	if = {
		limit = {
			scope:actor = {
				government_has_flag = government_is_clan
				house ?= scope:recipient.house
			}
		}
		add = 100

		# Personality
		if = {
			limit = { ai_wants_high_unity = yes }
			multiply = 1.2
		}
		else_if = {
			limit = { ai_wants_low_unity = yes }
			multiply = 0.8
		}

		# House Head Related
		if = {
			limit = {
				house ?= {
					OR = {
						has_house_unity_stage = antagonistic
						has_house_unity_stage = competitive
						has_house_unity_stage = impassive
					}
				}
				NOR = {
					house.house_head = top_liege
					house.house_head = liege
				}
			}
			multiply = 1.2
		}

		# Struggle traits
		if = {
			limit = { has_trait = fp3_struggle_supporter }
			multiply = 1.5
		}
		else_if = {
			limit = { has_trait = fp3_struggle_detractor }
			multiply = 0.5
		}

		# Current Unity Stage
		if = {
			limit = {
				house ?= {
					has_house_unity_stage = antagonistic
				}
			}
			multiply = 1.2
		}
		else_if = {
			limit = {
				house ?= {
					has_house_unity_stage = competitive
				}
			}
			multiply = 1.1
		}
		else_if = {
			limit = {
				house ?= {
					has_house_unity_stage = friendly
				}
			}
			multiply = 0.9
		}
		else_if = {
			limit = {
				house ?= {
					has_house_unity_stage = harmonious
				}
			}
			multiply = {
				value = {
					add = 0.8
					if = {
						limit = {
							house ?= { house_unity_value >= very_high_house_unity }
						}
						multiply = 0.5
					}
				}
			}
		}
	}

}
