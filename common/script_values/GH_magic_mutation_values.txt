child_mutation_severity_value = { # Modifies how likely a child is to get a severe mutation and how many they might be born with
	value = 0
	if = {
		limit = { exists = scope:mother }
		add = scope:mother.child_magic_defect_chance_value
	}
	else_if = {
		limit = { exists = mother }
		add = mother.child_magic_defect_chance_value
	}

	if = {
		limit = { exists = scope:real_father }
		add = scope:real_father.child_magic_defect_chance_value
	}
	else_if = {
		limit = { exists = scope:father }
		add = scope:father.child_magic_defect_chance_value
	}
	else_if = {
		limit = { exists = father }
		add = father.child_magic_defect_chance_value
	}
}
