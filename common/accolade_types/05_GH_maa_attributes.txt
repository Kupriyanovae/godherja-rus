charioteer_attribute = { # Charioteer
	adjective = accolade_chariot_riding
	noun = accolade_charioteer
	icon = "gfx/interface/icons/knight_badge/icons/accolade_trait_wheel_green.dds"
	portrait_pose = "threatening"
	accolade_categories = { men_at_arms army_bonus militant equestrian forceful leader aggressive cavalry light_fighter agile }

	squire_perfect_fit = {
		charioteer_attribute_trigger = yes
	}
	squire_acceptable_fit = {
		charioteer_attribute_acceptable_trigger = yes
	}

	weight = {
		value = accolade_uninteresting_base_value
		scope:owner ?= {
			if = {
				limit = {
					number_maa_regiments_of_base_type = {
						type = chariot_cavalry
						value >= 1
					}
					culture ?= { has_innovation = innovation_war_chariots }
				}
				add = @has_relevant_maa_weight_up_value
				if = {
					limit = {
						number_maa_regiments_of_base_type = {
							type = chariot_cavalry
							value >= @number_maa_regiments_increment_1
						}
					}
					add = @has_relevant_maa_weight_up_value
				}
				if = {
					limit = {
						number_maa_regiments_of_base_type = {
							type = chariot_cavalry
							value >= @number_maa_regiments_increment_2
						}
					}
					add = @has_relevant_maa_weight_up_value
				}
				if = {
					limit = {
						number_maa_regiments_of_base_type = {
							type = chariot_cavalry
							value >= @number_maa_regiments_increment_3
						}
					}
					add = @has_relevant_maa_weight_up_value
				}
				if = {
					limit = {
						number_maa_regiments_of_base_type = {
							type = chariot_cavalry
							value >= @number_maa_regiments_increment_4
						}
					}
					add = @has_relevant_maa_weight_up_value
				}
				if = {
					limit = { root.culture ?= { culture_specializes_in_chariot_cavalry_maa = yes } }
					add = @respected_culture_for_maa_type_value
				}
				if = {
					limit = { root.culture ?= { GH_culture_shuns_chariot_cavalry_maa = yes } }
					add = @disinterested_culture_for_maa_type_value
				}
				add = accolade_weighted_up_in_earlier_eras_value
				add = {
					value = 1
					multiply = ai_boldness
				}
				add = {
					value = ai_energy
					multiply = 1
				}
				multiply = accolade_opinion_multiplier_value
			}
		}
	}

	ranks = {
		1 = {
			liege_modifier = {
				chariot_cavalry_max_size_add = @number_maa_regiments_increment_1
			}
			knight_army_modifier = {
				chariot_cavalry_damage_mult = @accolade_maa_damage_1
				chariot_cavalry_toughness_mult = @accolade_maa_toughness_1
				chariot_cavalry_pursuit_mult = @accolade_maa_pursuit_1
				movement_speed = @chariot_attr_movement_speed_1
			}
		}
		2 = {
			liege_modifier = {
				chariot_cavalry_maintenance_mult = @accolade_maa_maint_1
				chariot_cavalry_max_size_add = @number_maa_regiments_increment_2
			}
			knight_army_modifier = {
				chariot_cavalry_damage_mult = @accolade_maa_damage_4
				chariot_cavalry_toughness_mult = @accolade_maa_toughness_2
				chariot_cavalry_pursuit_mult = @accolade_maa_pursuit_2
				movement_speed = @chariot_attr_movement_speed_2
			}
			men_at_arms = { accolade_maa_charioteers }
		}
		3 = {

				chariot_cavalry_maintenance_mult = @accolade_maa_maint_4
