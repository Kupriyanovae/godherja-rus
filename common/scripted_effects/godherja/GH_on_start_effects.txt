gh_on_start_ruler_designer_government_effect = {
	if = {
		limit = { exists = title:c_oraispol.var:GH_ruler_designer_government_override }
		switch = {
			trigger = title:c_oraispol.var:GH_ruler_designer_government_override
			flag:feudal_government_flag = { change_government = feudal_government }
			flag:republic_playable_government_flag = { change_government = republic_playable_government }
			flag:theocracy_government_flag = { change_government = theocracy_government }
			flag:landed_order_government_flag = { change_government = landed_order_government }
			flag:military_unit_government_flag = { change_government = military_unit_government }
			flag:magocratic_government_flag = { change_government = magocratic_government }
			flag:magocratic_republic_flag = { change_government = magocratic_republic }
			flag:pirate_government_flag = { change_government = pirate_government }
		}
		title:c_oraispol = { remove_variable = GH_ruler_designer_government_override }
	}
}

gh_set_nonexistent_faith_effect = {
	# Faiths that do not exist except for player intervention - cannot be converted to unless formed through decision.
	if = {
		limit = { $LORE$ = no } # Indicates the title was not created in lore (before 1298, at least)
		faith:$FAITH$ = { set_variable = { name = block_conversion_till_decision_taken } }
	}
	# Faith does not exist before a certain date but do afterwards - cannot be converted to unless formed through events or playing after formation date
	else = {
		limit = {
			game_start_date < $DATE$
		}
		faith:$FAITH$ = { set_variable = { name = block_conversion_till_nebulous_circumstances } }
	}
}

gh_on_start_every_character_effect = {
	every_living_character = {

		## Opening Events ##
		if = {
			limit = { has_opening_event = yes }
			add_trait = setup_flag_opening_event
		}
		if = {
			limit = { has_story_content = yes }
			add_trait = setup_flag_story_content
		}
		if = {
			limit = {
				is_ruler = yes
				OR = {
					has_trait = setup_flag_opening_event
					has_trait = setup_flag_story_content
				}
			}
			save_scope_as = lore_char
			title:c_oraispol = {
				add_to_variable_list = {
					name = lore_character_list
					target = scope:lore_char
				}
			}
		}

		## Lifepaths ##
		if = {
			limit = {
				faith = { 
					OR = {
						has_variable = faith_has_lifepaths
						has_doctrine = tenet_lifepath_religion_syncretism 
					}
				}
			}
			if = {
				limit = {
					is_ruler = yes
				}
				populate_with_wayfathers = yes
			}
			if = {
				limit = {
					age >= 13
				}
				assign_all_lifepath_trait = yes
			}
		}

		## Wardenism ##
		if = {
			limit = {
				should_adopt_legal_tradition_trigger = yes
				NOT = { has_trait = lifestyle_jurist }
			}
			random_list = {
				70 = {}
				15 = {
					add_trait = lifestyle_jurist
					assign_character_legal_tradition_effect = yes
				}
				10 = {
					add_trait = lifestyle_jurist
					add_trait_xp = {
						trait = lifestyle_jurist
						value = 25
					}
					assign_character_legal_tradition_effect = yes
				}
				5 = {
					add_trait = lifestyle_jurist
					add_trait_xp = {
						trait = lifestyle_jurist
						value = 50
					}
					assign_character_legal_tradition_effect = yes
				}
			}
		}

		## Lifestyles traits ##
		gh_clean_historical_lifestyles_effect = yes # Replace historical lifestyle traits with correct ones

		## Republic Setup ##
		if = {
			limit = {
				OR ={
					government_has_flag = government_is_administrative_republic
					government_has_flag = government_is_judicial_republic
				}
				government_allows = administrative
				government_has_flag = government_has_powerful_families
			}
			if = {
				limit = {
					highest_held_title_tier >= tier_county
					government_has_flag = government_is_administrative_republic
				}
				trigger_event = {
					id = gh_realm_maintenance.0036
					days = { 2 5 }
				}
			}
			save_scope_as = ruler
			every_vassal = { # Makes sure all the vassals have NFs if not defined in history
				if = {
					limit = {
						highest_held_title_tier >= tier_county
						is_lowborn = no
						is_house_head = yes
						NOT = { capital_province = { has_holding_type = church_holding } }
						government_has_flag = government_has_powerful_families
						government_allows = administrative
						OR = {
							is_married = no
							AND = {
								is_married = yes
								is_female = yes
								matrilinear_marriage = yes
							}
							AND = {
								is_married = yes
								is_female = no
								matrilinear_marriage = no
							}
						}
						NOR = {
							any_held_title = { is_noble_family_title = yes }
							house = {
								any_house_member = {
									any_held_title = { is_noble_family_title = yes }
								}
							}
						}
					}
					create_noble_family_effect = { GOVERNMENT_GIVER = this }
					domicile ?= { set_up_domicile_estate_effect = yes }
				}
			}
		}

		## Konijkmener Setup ##
		if = {
			limit = { has_title = title:d_kanzelerij_monarchy }
			save_scope_as = konijk_monarch
			add_gold = massive_gold_value
			house = {
				every_house_member = {
					limit = {
						is_ruler = no
						is_married = no
						NOT = { liege = scope:konijk_monarch }
					}
					save_scope_as = konijk_family_member
					scope:konijk_monarch = { add_courtier = scope:konijk_family_member }
				}
			}
			dynasty = {
				while = {
					limit = {
						dynasty_prestige_level < high_dynasty_prestige_level 
					}
					add_dynasty_prestige_level = 1
				}
			}
		}
	}
}

gh_historical_lifestyle_effect = { # For replacing historical lifestyle traits that do not account for tracks
	if = { # 999 indicates no XP
		limit = { $XP$ = 999 }
		if = {
			limit = { has_trait = gh_historical_$HISTORICAL$ }
			remove_trait = gh_historical_$HISTORICAL$
			add_trait = $TRAIT$
		}
	}
	else = { # Has XP but no track
		if = {
			limit = { has_trait = gh_historical_$HISTORICAL$ }
			remove_trait = gh_historical_$HISTORICAL$
			add_trait = $TRAIT$
			add_trait_xp = {
				trait = $TRAIT$
				value = $XP$
			}
		}
	}
}
gh_historical_lifestyle_with_track_effect = { # Likewise for tracked traits
	if = { # Has both
		limit = { has_trait = gh_historical_track_$HISTORICAL$ }
		remove_trait = gh_historical_track_$HISTORICAL$
		add_trait = $TRAIT$
		add_trait_xp = {
			trait = $TRAIT$
			track = $TRACK$
			value = $XP$
		}
	}
}

gh_on_start_every_county_effect = {
	every_county = {
		limit = { is_landless_type_title = no }
		# Setup Metros
		gh_initialize_district_types_effect = yes
		if = { # Runs on game start to add building slots to metropoli and add in buildings that would otherwise refuse to load due to previous build cap
			limit = {
				title_province = {	is_any_metropolis_or_ruined_district_trigger = yes }
			}
			every_county_province = {
				if = {
					limit = { # County capitals get two additional building slots
						is_county_capital = yes
					}
					add_province_modifier = extra_building_slot
					add_district_list_effect = yes
					add_random_district_effect = yes
					add_random_district_effect = yes
				}
			}
		}

		# Setup coastal counties
		if = {
			limit = { is_coastal_county = yes }
			save_scope_as = coastal_county
			add_to_global_variable_list = {
				name = coastal_counties
				target = this
			}
		}

		# Setup wastelands
		if = {
			limit = {
				holder = {
					faith = faith:wasteland_faith
					culture = culture:wasteland
					has_trait = gh_wasteland
				}
			}
			save_temporary_scope_as = ruined_county
			spawn_ruin = { COUNTY = scope:ruined_county }
		}

		# Setup Lichdoms
		if = {
			limit = {
				gh_special_province_lichdom_trigger = yes
			}
			random_list = {
				100 = {
				    modifier = {
				    	add = {
			    			subtract = necropolis_spawn_chance
			    		}
				    }
				}
				0 = {
				    modifier = {
				    	add = necropolis_spawn_chance
				    }
					spawn_special_province_necropolis = yes
				}
			}
		}

		# Clear up wasteland issues
		gh_wasteland_and_ruin_check_effect = yes
		gh_bordering_fog_dynamic_effect = yes
	}
}
