set_artifact_as_magical = {
	scope:newly_created_artifact = {
		save_scope_as = this_artifact
	}
	scope:this_artifact = {
		set_variable = {
			name = magical_artifact
			value = yes
		}
		random_list = {
			10 = { add_artifact_aspect_modifier_effect = { NAME = blood VALUE = 1 } }
			10 = { add_artifact_aspect_modifier_effect = { NAME = water VALUE = 1 } }

artifact_magical_infusion_effect = { # This is not for giving artifacts magic, this is for magic features
	if = {
		limit = {
			scope:newly_created_artifact = {
				OR = {
					AND = {
						exists = var:quality
						var:quality > 50
					}
					OR = {
						rarity = famed
						rarity = illustrious
					}
				}

			value = $MIN_YEAR$

					min = 0

					}
				}
			}

	set_variable = {
		name = $NAME$_upper_year
		value = {
			value = var:$NAME$_lower_year
			add = {
				integer_range = {
					min = $MIN_LENGTH$
					max = $MAX_LENGTH$
				}
			}
		}
	}

	$POTTER$ = { save_scope_as = potter }
	$LOCATION$ = { save_scope_as = location }
	if = { #Any existing dummy character gets turned into the potter
		limit = {
			exists = scope:dummy_gender
			scope:potter = scope:dummy_gender
		}
		clear_saved_scope = potter
		clear_saved_scope = dummy_gender
	}
	hidden_effect_new_object = {
		# Get artifact quality, wealth, materials, and decorations
		get_artifact_quality_effect = yes
		get_artifact_wealth_effect = yes

		# Create the artifact
		if = {
			limit = { exists = scope:adventurer }
			create_artifact = {
				name = artifact_vessel_earthware_name
				description = placeholder
				visuals = vessel
				type = vessel_earthware
				modifier = artifact_placeholder_modifier
				wealth = scope:wealth
				quality = scope:quality
				save_scope_as = newly_created_artifact
				history = {
					location = scope:location
					actor = scope:adventurer
					recipient = root
					type = discovered
				}
			}
		}
		else_if = {
			limit = {
				scope:host ?= $OWNER$
				scope:activity ?= { has_activity_type = activity_tournament }
			}
			create_artifact = {
				name = artifact_vessel_earthware_name
				creator = scope:host
				description = placeholder # Will be updated in the post-creation effects after decorations are assigned.
				visuals = vessel
				type = vessel_earthware
				modifier = artifact_placeholder_modifier
				wealth = scope:wealth
				quality = scope:quality
				history = {
					type = prize_created
					recipient = scope:host
				}
				save_scope_as = newly_created_artifact
			}
		}
		else_if = {
			limit = { exists = scope:potter }
			create_artifact = {
				name = artifact_vessel_earthware_name
				creator = scope:potter
				description = placeholder # Will be updated in the post-creation effects after decorations are assigned.
				visuals = vessel
				type = vessel_earthware
				modifier = artifact_placeholder_modifier
				wealth = scope:wealth
				quality = scope:quality
				save_scope_as = newly_created_artifact
			}
		}
		else = {
			create_artifact = {
				name = artifact_vessel_earthware_name
				description = placeholder # Will be updated in the post-creation effects after decorations are assigned.
				visuals = vessel
				type = vessel_earthware
				modifier = artifact_placeholder_modifier
				wealth = scope:wealth
				quality = scope:quality
				save_scope_as = newly_created_artifact
				history = { type = created_before_history }
			}
		}
		scope:newly_created_artifact = {
			# Finish generating decorations and update description
			get_artifact_feature_references_effect = yes
			set_artifact_description = artifact_vessel_earthware_description
			# Add artifact modifiers
			add_scaled_artifact_modifier_grandeur_small_effect = yes
			add_scaled_artifact_modifier_minor_prestige_effect = yes
			if = {
				limit = {
					OR = {
						rarity = famed
						rarity = illustrious
					}
				}
				add_scaled_artifact_modifier_majesty_effect = yes
			}
			set_variable = {
				name = quality
				value = scope:quality
			}
			set_variable = {
				name = wealth
				value = scope:wealth
			}
		}
	}
}

					BASE_SCOPE = scope:smith
					NO_TYPE = no
				}
				scope:smith = {
					var:artifact_weapon_type = { save_scope_as = weapon_type }
					remove_variable = artifact_weapon_type
				}
			}
			else = {
				set_weapon_artifact_type_effect = {
					BASE_SCOPE = scope:owner
					NO_TYPE = no
				}
				scope:owner = {
					var:artifact_weapon_type = { save_scope_as = weapon_type }
					remove_variable = artifact_weapon_type
				}
			}
		}
		else = {
			scope:weapon_type_is_set = { save_scope_as = weapon_type }
		}
		# Get artifact quality and wealth.
		random_list = {
			4 = { set_artifact_rarity_famed = yes }
			1 = { set_artifact_rarity_illustrious = yes }
		}

		# Get the type of weapon to forge. Defaults to sword if an invalid type is provided.
		# GH TODO: Add other weapon types once we get the models for them
		if = {
			limit = { always = yes }
			if = {
				limit = {
					scope:host ?= $OWNER$
					scope:activity ?= { has_activity_type = activity_tournament }
					NOT = { $OWNER$ = { has_character_flag = event_4500_happening } }
				}
				random_list = {
					1 = {
						tournament_contest_weapon_prize_effect = { VISUALS = longsword TYPE = sword }
					}
					1 = {
						tournament_contest_weapon_prize_effect = { VISUALS = sword TYPE = sword }
					}
				}
			}
			else = {
				if = {
					limit = { scope:newly_created_artifact = { rarity = illustrious } }
					create_artifact = {
						name = artifact_sword_name
						description = placeholder # Will be re-generated once decorations are added in the post-creation effects.
						visuals = gh_aelfir_sword_gold_visuals
						type = sword
						modifier = artifact_prowess_1_modifier
						wealth = scope:wealth
						quality = scope:quality
						save_scope_as = newly_created_artifact
						history = { type = created_before_history }
					}
				}
				else = {
					create_artifact = {
						name = artifact_sword_name
						description = placeholder # Will be re-generated once decorations are added in the post-creation effects.
						visuals = gh_aelfir_sword_bare_visuals
						type = sword
						modifier = artifact_prowess_1_modifier
						wealth = scope:wealth
						quality = scope:quality
						save_scope_as = newly_created_artifact
						history = { type = created_before_history }
					}
				}
			}

			scope:newly_created_artifact = {
				set_artifact_feature = GH_blade_metal_unknown_dark
				get_sword_decoration_effect = yes
				get_artifact_feature_references_effect = yes
				set_artifact_description = artifact_sword_description
			}
		}

		scope:newly_created_artifact = {
			add_artifact_history = {
				type = discovered
				location = scope:location
				actor = scope:smith
				recipient = scope:owner
			}
			# Prowess from weapon rarity
			add_scaled_artifact_modifier_prowess_effect = yes

			if = {
				limit = {
					OR = {
						rarity = famed
						rarity = illustrious
					}
					OR = {
						has_artifact_feature_group = blade_decoration
						has_artifact_feature_group = hilt_decoration
						has_artifact_feature_group = shaft_decoration
						has_artifact_feature_group = head_decoration
						scope:weapon_type = flag:artifact_weapon_type_dagger
					}
				}
				if = {
					limit = { scope:weapon_type = flag:artifact_weapon_type_dagger }
					add_2_scaled_artifact_modifier_intrigue_effect = yes
				}
				else = {
					add_2_scaled_artifact_modifier_combat_effect = yes
				}
			}
			else_if = {
				limit = {
					OR = {
						rarity = common
						rarity = masterwork
					}
					OR = {
						has_artifact_feature_group = blade_decoration
						has_artifact_feature_group = hilt_decoration
						has_artifact_feature_group = shaft_decoration
						has_artifact_feature_group = head_decoration
						scope:weapon_type = flag:artifact_weapon_type_dagger
					}
				}
				if = {
					limit = { scope:weapon_type = flag:artifact_weapon_type_dagger }
					add_scaled_artifact_modifier_intrigue_effect = yes
				}
				else = {
					add_scaled_artifact_modifier_combat_effect = yes
				}
			}
			else_if = {
				limit = {
					OR = {
						rarity = famed
						rarity = illustrious
					}
					# No decorations
				}
				if = {
					limit = { scope:weapon_type = flag:artifact_weapon_type_dagger }
					add_scaled_artifact_modifier_intrigue_effect = yes
					add_artifact_modifier = artifact_prowess_no_decorations_high_tier_modifier
				}
				else = {
					add_scaled_artifact_modifier_combat_effect = yes
					add_artifact_modifier = artifact_prowess_no_decorations_high_tier_modifier
				}
			}
			else_if = {
				limit = {
					rarity = masterwork
					# No decorations
				}
				if = {
					limit = { scope:weapon_type = flag:artifact_weapon_type_dagger }
					add_scaled_artifact_modifier_intrigue_effect = yes
					add_artifact_modifier = artifact_prowess_no_decorations_modifier
				}
				else = {
					add_artifact_modifier = artifact_prowess_no_decorations_medium_tier_modifier
				}
			}
			else = {
				add_artifact_modifier = artifact_prowess_no_decorations_modifier
			}

			# Save the quality and wealth of this artifact in case we need to reference it later.
			set_variable = {
				name = quality
				value = scope:quality
			}
			set_variable = {
				name = wealth
				value = scope:wealth
			}
		}
	}
}

add_artifact_aspect_modifier_effect = {
	if = {
		limit = { NOT = { $VALUE$ = 0 } }
		custom_description = {
			text = gain_$NAME$_artifact_modifier
			value = $VALUE$
			
			add_to_variable_list = {
				name = artifact_aspect_modifier_list
				target = flag:$NAME$
			}
			if = {
				limit = { 
					NOT = { exists = var:$NAME$ }
				}
				set_variable = {
					name = $NAME$
					value = $VALUE$
				}
			}
			else = {
				change_variable = {
					name = $NAME$
					add = $VALUE$
				}
			}
			
			if = {	# GH TODO: Make it calculate if it's the highest aspect instead
				limit = { NOT = { exists = var:primary_aspect_aesthetics } }
				set_variable = {
					name = primary_aspect_aesthetics
					value = flag:$NAME$
				}
				set_variable = {
					name = secondary_aspect_aesthetics
					value = flag:$NAME$
				}
			}
			else_if = {
				limit = {
					NOT = {
						var:primary_aspect_aesthetics = flag:$NAME$
					}
				}
				set_variable = {
					name = secondary_aspect_aesthetics
					value = flag:$NAME$
				}
			}
		}
	}
}

		}

		if = {
			limit = { NOT = { exists = scope:location } }

		create_artifact_book_effect = {

			SET_TOPIC = flag:arcana_magic_school

			}

			set_variable = {
				name = author
				value = scope:author
			}

	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	
	save_scope_value_as = {
		name = book_uses_template
		value = flag:religious
	}
	
	hidden_effect_new_object = {
		random_list = {
			7 = { set_artifact_rarity_masterwork = yes }
			2 = { set_artifact_rarity_famed = yes }
			1 = { set_artifact_rarity_illustrious = yes }
		}

		# Create the artifact
		create_artifact_book_effect = {
			OWNER = scope:owner
			CREATOR = scope:owner
			SET_SUBJECT = flag:learning
			SET_TOPIC = flag:religious
		}
		
		scope:newly_created_artifact = { 
			set_variable = {
				name = faith_override
				value = $FAITH$
			}
			# Some faiths have a very large scripture
			if = {
				limit = {
					OR = {
						$FAITH$.religion = religion:wardenism_religion
					}
				}
				set_artifact_feature = book_topic_learning_religious_big
				# Massive books are hard to come by, even without fancy illustrations, covers or calligraphy
				if = {
					limit = {
						OR = {
							rarity = common
							rarity = masterwork
						}
					}
					set_artifact_rarity = famed
				}
			}
			set_artifact_feature = book_title_learning_religious
			# Generate the updated - final description for the book
			set_artifact_description = artifact_book_description
			# Generate the updated title - final title for the book
			set_artifact_name = artifact_book_title
		}
	}
}

# Misc
create_artifact_skull_effect = {
	$OWNER$ = { save_scope_as = owner }
	$DESKULLED$ = { save_scope_as = killed }

	save_scope_value_as = {
		name = quality
		value = {
			value = 10
			add = scope:killed.prestige_level
			add = scope:killed.piety_level
			if = { # Boost for notable skull donors
				limit = { scope:killed = { is_lowborn = no } }
				add = {
					value = scope:killed.dynasty.dynasty_prestige_level
					multiply = 5
				}
			}
			multiply = scope:killed.highest_held_title_tier
			if = { # Boost for battle skulls
				limit = { exists = scope:combat_location }
				multiply = 1.25
			}
			if = { # Boost for exceptional individuals
				limit = { scope:killed = { has_education_rank_5_trigger = yes } }
				multiply = 1.25
			}
			round = yes
		}
	}
	save_scope_value_as = {
		name = wealth
		value = scope:quality
	}

	create_artifact = {
		name = placeholder
		description = placeholder
		type = skull
		visuals = human_skull
		wealth = scope:wealth
		quality = scope:quality
		modifier = artifact_placeholder_modifier
		history = { # A bit cheeky
			type = created
			actor = scope:killed
			recipient = scope:owner
			location = scope:owner.location
		}
		save_scope_as = newly_created_artifact
	}

	hidden_effect_new_object = {
		scope:newly_created_artifact = {
			if = { # Get appropriate name format
				limit = { scope:killed = { is_ruler = yes } }
				set_artifact_name = artifact_skull_name_ruler
			}
			else = { set_artifact_name = artifact_skull_name_not_ruler }
			set_artifact_feature_group = skull_condition
			set_artifact_feature_group = skull_circumstance
			switch = {
				trigger = has_artifact_feature
				skull_circumstance_battle = {
					set_variable = {
						name = battle_location
						value = scope:combat_location
					}
					set_artifact_feature_group = skull_circumstance_battle_description
				}
				skull_circumstance_war = {
					set_variable = {
						name = battle_location
						value = scope:combat_location
					}
					set_variable = {
						name = battle_war
						value = scope:combat_war
					}
					set_artifact_feature_group = skull_circumstance_battle_description
				}
				skull_circumstance_execution = { set_artifact_feature_group = skull_circumstance_execution_description }
				skull_circumstance_duel = {}
			}
			set_artifact_description = artifact_skull_description
			add_scaled_artifact_modifier_prestige_effect = yes
			switch = {
				trigger = rarity
				common = { add_artifact_modifier = artifact_dread_baseline_add_1_modifier }
				masterwork = {
					add_artifact_modifier = artifact_dread_baseline_add_2_modifier
					random = {
						chance = 50
						random_list = {
							1 = { add_scaled_artifact_modifier_majesty_effect = yes }
							1 = { add_scaled_artifact_modifier_dynasty_prestige_effect = yes }
						}
					}
				}
				famed = {
					add_artifact_modifier = artifact_dread_baseline_add_3_modifier
					random_list = {
						1 = { add_scaled_artifact_modifier_majesty_effect = yes }
						1 = { add_scaled_artifact_modifier_dynasty_prestige_effect = yes }
					}
				}
				illustrious = {
					add_artifact_modifier = artifact_dread_baseline_add_4_modifier
					add_scaled_artifact_modifier_majesty_effect = yes
					add_scaled_artifact_modifier_dynasty_prestige_effect = yes
				}
			}
			set_variable = artifact_skull
			set_variable = {
				name = skull_donor
				value = scope:killed
			}
			set_artifact_description = artifact_skull_description_epitaph
			set_max_durability = 50 # Skulls are fragile
			remove_artifact_modifier = artifact_placeholder_modifier
		}
		scope:killed.house ?= { add_house_artifact_claim = scope:newly_created_artifact }
	}
}

		name = allow_no_type_option
		value = $NO_TYPE$
	}
	$BASE_SCOPE$ = {
		if = { # If no Artifact type yet exists
			limit = { NOT = { exists = var:artifact_glassware_type } }
			random_list = {
				10 = { # Beaker
					modifier = {
						factor = 2
						has_trait = content
					}
					modifier = {
						factor = 1.2
						glassware_inspiration_difficulty_medium_check = yes
					}
					modifier = {
						factor = 0.5
						glassware_inspiration_difficulty_hard_check = yes
					}
					set_variable = {
						name = artifact_glassware_type
						value = flag:artifact_glassware_type_beaker
					}
				}
				50 = { # You decide
					trigger = { scope:allow_no_type_option = yes }
					modifier = {
						add = 30
						has_personality_submissive_trigger = yes
					}
					modifier = {
						add = 10000
						has_character_flag = local_artisan
					}
					ai_value_modifier = {
						ai_boldness = -0.25
						ai_energy = -0.25
					}
				}
			}
		}
	}
}

create_artifact_glassware_effect = { # Create a Glassware Artifact
	$OWNER$ = { save_scope_as = owner }
	$CREATOR$ = { save_scope_as = glassmith }
	save_temporary_scope_value_as = { # Save the predetermined Glassware Artifact type or indicate that no such thing exists
		name = glassware_type_is_set
		value = $SET_GLASSWARE_TYPE$
	}
	if = { # Clear dummy glassmith gender if such exists
		limit = {
			exists = scope:dummy_gender
			scope:glassmith = scope:dummy_gender
		}
		clear_saved_scope = dummy_gender
		clear_saved_scope = glassmith
	}
	hidden_effect_new_object = {
		if = { # Pick a Glassware Artifact type if it does not exist yet
			limit = { scope:glassware_type_is_set = flag:no }
			if = { # Base it on the glassmith Character if such exists
				limit = { exists = scope:glassmith }
				set_glassware_artifact_type_effect = {
					BASE_SCOPE = scope:glassmith
					NO_TYPE = no
				}
				scope:glassmith = { # Save the chosen Artifact type
					var:artifact_glassware_type = { save_scope_as = glassware_type }
					remove_variable = artifact_glassware_type
				}
			}
			else = { # Base it on the owner Character instead
				set_glassware_artifact_type_effect = {
					BASE_SCOPE = scope:owner
					NO_TYPE = no
				}
				scope:owner = { # Save the chosen Artifact type
					var:artifact_glassware_type = { save_scope_as = glassware_type }
					remove_variable = artifact_glassware_type
				}
			}
		}
		else = { scope:glassware_type_is_set = { save_scope_as = glassware_type } } # Save the existing Artifact type
		switch = { # Create the final Artifact type
			trigger = scope:glassware_type
			#flag:artifact_glassware_type_beads = { create_artifact_glass_beads_effect = yes }
			#flag:artifact_glassware_type_flask = { create_artifact_glass_flask_effect = yes }
			flag:artifact_glassware_type_beaker = { create_artifact_beaker_effect = yes }
			#flag:artifact_glassware_type_ewer = { create_artifact_glass_ewer_effect = yes }
			#flag:artifact_glassware_type_vase = { create_artifact_glass_vase_effect = yes }
			#flag:artifact_glassware_type_sculpture = { create_artifact_glass_sculpture_effect = yes }
		}
	}
}

set_glass_properties_effect = { # Pick universal Glass Artifact Features
	if = { # Pick a pre-determined Glass Composition from Personal Inspiration if such exists
		limit = { scope:owner.var:personal_inspiration_story ?= { has_variable = chosen_glass_composition } }
		scope:owner.var:personal_inspiration_story = {
			switch = {
				trigger = var:chosen_glass_composition
				flag:forest_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_forest_glass } }
				flag:smalt_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_smalt_glass } }
				flag:antimony_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_antimony_glass } }
				flag:copper_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_copper_glass } }
				flag:magnesia_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_magnesia_glass } }
				flag:lead_glass = { scope:newly_created_artifact = { set_artifact_feature = glass_composition_lead_glass } }
			}
		}
	}
	else = { set_artifact_feature_group = glass_composition } # Or pick a random one otherwise
	if = { # All Glass Composition Features sans sintered quartz get a Glass Transparency Feature
		limit = { NOT = { has_artifact_feature = glass_composition_sintered_quartz } }
		set_artifact_feature_group = glass_transparency
	}
	if = { # All Glass Transparency Features sans complete transparency get a Glass Color Feature
		limit = { NOT = { has_artifact_feature = glass_transparency_transparent } }
		set_artifact_feature_group = glass_color
	}
}

	}
	add_trait_xp = {
		trait = gh_lifestyle_glassmith
		value = glassmith_inspiration_experience_gain
	}
}

populate_beaker_decorations_effect = { # Pick decorative Features for the Glass Beaker Artifact
	while = {
		count = { $MIN_FEATURE_COUNT$ $MAX_FEATURE_COUNT$ }
		random_list = {
			1 = { # Decoration
				trigger = { NOT = { has_artifact_feature_group = beaker_decoration } }
				set_artifact_feature_group = beaker_decoration
			}
			1 = { # Pattern
				trigger = { NOT = { has_artifact_feature_group = decoration_pattern } }
				set_artifact_feature_group = decoration_pattern
			}
			1 = { # Inlay
				trigger = { NOT = { has_artifact_feature_group = decoration_material_inlay } }
				set_artifact_feature_group = decoration_material_inlay
			}
			1 = { # Wire
				trigger = { NOT = { has_artifact_feature_group = decoration_material_wire } }
				set_artifact_feature_group = decoration_material_wire
			}
			1 = { # Gem
				trigger = { NOT = { has_artifact_feature_group = decoration_material_gem } }
				set_artifact_feature_group = decoration_material_gem
			}
		}
	}
}

create_artifact_beaker_effect = { # Create a Glass Beaker Artifact
	get_artifact_quality_effect = yes
	get_artifact_wealth_effect = yes
	if = { # Adventurer inspiration
		limit = { exists = scope:adventurer }
		create_artifact = {
			name = artifact_beaker_name
			description = placeholder
			visuals = beaker
			type = beaker
			modifier = artifact_placeholder_modifier
			wealth = scope:wealth
			quality = scope:quality
			save_scope_as = newly_created_artifact
			history = {
				location = scope:location
				actor = scope:adventurer
				recipient = root
				type = discovered
			}
		}
	}
	else_if = { # Tournament prize
		limit = {
			scope:host ?= scope:owner
			scope:activity ?= { has_activity_type = activity_tournament }
		}
		create_artifact = {
			name = artifact_beaker_name
			creator = scope:host
			description = placeholder
			visuals = beaker
			type = beaker
			modifier = artifact_placeholder_modifier
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = prize_created
				recipient = scope:host
			}
			save_scope_as = newly_created_artifact
		}
	}
	else_if = { # Glassware inspiration
		limit = { exists = scope:glassmith }
		create_artifact = {
			name = artifact_beaker_name
			creator = scope:glassmith
			description = placeholder
			visuals = beaker
			type = beaker
			wealth = scope:wealth
			quality = scope:quality
			modifier = artifact_placeholder_modifier
			save_scope_as = newly_created_artifact
		}
	}
	else = { # Fallback
		create_artifact = {
			name = artifact_beaker_name
			description = placeholder
			visuals = beaker
			type = beaker
			wealth = scope:wealth
			quality = scope:quality
			modifier = artifact_placeholder_modifier
			save_scope_as = newly_created_artifact
			history = { type = created_before_history }
		}
	}

	scope:newly_created_artifact = {
		set_glass_properties_effect = yes
		set_artifact_feature_group = beaker_shape
		random_list = { # Pick a number of decorative Features based on complexity
			40 = {
				trigger = { ep1_extravagant_artifact_decorations_trigger = yes }
				populate_beaker_decorations_effect = { MIN_FEATURE_COUNT = 2 MAX_FEATURE_COUNT = 4 }
			}
			30 = {
				trigger = {	ep1_advanced_artifact_decorations_trigger = yes }
				populate_beaker_decorations_effect = { MIN_FEATURE_COUNT = 1 MAX_FEATURE_COUNT = 3 }
			}
			20 = {
				trigger = {	ep1_simple_artifact_decorations_trigger = yes }
				populate_beaker_decorations_effect = { MIN_FEATURE_COUNT = 1 MAX_FEATURE_COUNT = 2 }
			}
			10 = { trigger = { ep1_no_artifact_decorations_trigger = yes } }
		}

		switch = { # Pick a number of Artifact Modifiers based on Rarity
			trigger = rarity
			common = {
				random_list = {
					1 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 2 } }
					3 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 1 } }
					2 = {}
				}
			}
			masterwork = {
				random_list = {
					1 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 2 } }
					1 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 1 } }
				}
			}
			famed = {
				random_list = {
					1 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 3 } }
					4 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 2 } }
				}
			}
			illustrious = {
				random_list = {
					1 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 3 } }
					2 = { populate_beaker_modifiers_effect = { MODIFIER_COUNT = 2 } }
				}
			}
		}

		if = { # Add Glass Composition-specific Artifact Modifiers
			limit = { artifact_rarity >= 4 }
			switch = {
				trigger = has_artifact_feature
				glass_composition_forest_glass = { add_scaled_artifact_modifier_rulership_effect = yes }
				glass_composition_smalt_glass = { add_scaled_artifact_modifier_majesty_effect = yes }
				glass_composition_antimony_glass = { add_scaled_artifact_modifier_intrigue_effect = yes }
				glass_composition_copper_glass = { add_scaled_artifact_modifier_combat_effect = yes }
				glass_composition_magnesia_glass = { add_scaled_artifact_modifier_scholarship_effect = yes }
				glass_composition_lead_glass = {
					random_list = {
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = air } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = water } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = blood } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = fire } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = metal } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = dream } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = stone } }
						1 = { add_scaled_artifact_modifier_aspect_effect = { ASPECT = death } }
					}
				}
			}
		}

		add_scaled_artifact_modifier_minor_prestige_effect = yes
		add_scaled_artifact_modifier_grandeur_small_effect = yes

		get_artifact_feature_references_effect = yes
		set_artifact_description = artifact_beaker_description
	}
}
