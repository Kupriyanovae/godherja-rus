reformer_convert_county_effect = {
	duel = {
		skill = learning
		value = $COUNTY$.development_level
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 3.5
				min = -49
			}
			desc = sjalvolki_reformation.conversion_success.tt
			$COUNTY$.holder = {
				send_interface_toast = {
					title = sjalvolki_reformation.conversion_success.tt
					left_icon = $COUNTY$.holder
					right_icon = scope:sjalvolki_reformer
					$COUNTY$ = {
						set_county_faith = faith:orispolj_rite
					}
					scope:sjalvolki_reformer = {
						add_prestige = {
							value = $COUNTY$.development_level
							multiply = 10
						}
						add_piety = {
							value = $COUNTY$.development_level
							multiply = 5
						}
					}
				}
			}
			set_local_variable = {
				name = notoriety_increase
				value = {
					value = $COUNTY$.development_level
					multiply = 0.1
					min = 1
				}
			}
			increase_reformer_notoriety_and_fervor_effect = { VALUE = local_var:notoriety_increase }
		}
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = -3.5
				min = -49
			}
			desc = sjalvolki_reformation.conversion_failure.tt
			send_interface_toast = {
				title = sjalvolki_reformation.conversion_failure.tt
				left_icon = $COUNTY$.holder
				right_icon = scope:sjalvolki_reformer
				scope:sjalvolki_reformer = {
					add_prestige = minor_prestige_loss
				}
			}
			increase_reformer_notoriety_and_fervor_effect = { VALUE = 1 }
		}
	}
}

increase_fervor_effect = {
	if = {
		limit = { exists = scope:sjalvolki_reformist_story.var:reform_fervor }
		scope:sjalvolki_reformist_story = {
			change_variable = {
				name = reform_fervor
				add = $VALUE$
			}
		}
	}
	else = {
		scope:sjalvolki_reformist_story = {
			set_variable = {
				name = reform_fervor
				value = $VALUE$
			}
		}
	}
}

increase_reformer_notoriety_and_fervor_effect = {
	scope:sjalvolki_reformer = {
		change_variable = {
			name = notoriety
			add = $VALUE$
		}
		if = {
			limit = { exists = scope:sjalvolki_reformist_story }
			if = {
				limit = { exists = scope:sjalvolki_reformist_story.var:reform_fervor }
				scope:sjalvolki_reformist_story = {
					change_variable = {
						name = reform_fervor
						add = {
							value = $VALUE$
							multiply = 0.5
						}
					}
				}
			}
			else = {
				scope:sjalvolki_reformist_story = {
					set_variable = {
						name = reform_fervor
						value = $VALUE$
					}
					change_variable = {
						name = reform_fervor
						multiply = 0.5
					}
				}
			}
		}
	}
}

form_the_solar_rite_decision_effects_scripted_effect = {
	scope:religious_leader = {
		add_character_flag = {
			flag = delay_player_faith_conversion_notification_event
			days = 1
		}
		faith:zaumiere_rite = {
			remove_variable = block_conversion_till_decision_taken
		}
		faith:kakarataki = {
			set_variable = { name = has_been_reformed }
		}
		set_character_faith_with_conversion = faith:zaumiere_rite
	}
	custom_tooltip = form_the_solar_rite_decision.county_conversions.tt
	hidden_effect = {
		primary_title = {
			every_in_de_facto_hierarchy = {
				limit = {
					tier = tier_county
					exists = scope:old_faith
					OR = {
						religion = scope:old_faith.religion
						religion = religion:sjalvolki_religion
					}
				}
				if = {
					limit = { culture = { has_cultural_pillar = heritage_karzarad } }
					random = {
						chance = 70
						set_county_faith = faith:zaumiere_rite
					}
				}
				else_if = {
					limit = { culture = { has_cultural_pillar = heritage_sjalvolki } }
					random = {
						chance = 50
						set_county_faith = faith:zaumiere_rite
					}
				}
				else = {
					random = {
						chance = 25
						set_county_faith = faith:zaumiere_rite
					}
				}
			}
		}
	}
	primary_title = {
		every_in_de_facto_hierarchy = {
			limit = {
				tier = tier_county
				faith = faith:zaumiere_rite
			}
			add_county_modifier = {
				modifier = zaumiere_jubilance_modifier
				years = 20
			}
		}
	}

	faith:zaumiere_rite = {
		change_fervor = {
			value = 100
			desc = fervor_gain_solar_rite_foundation
		}
	}
}

				limit = {

					religion = scope:old_faith.religion
				}
				#Goans have a high chance to flip.
				if = {
					limit = { 
						culture = {
							OR = {

						}

					random = {

				}

	}
