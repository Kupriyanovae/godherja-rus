# Anomalous Zones and Wizard's Towers
poi_special_buildings_arcana = {
	build_province_list = {
		every_special_building_province = {
			limit = {
				has_building_with_flag = { flag = travel_point_of_interest_arcana }
			}
			add_to_list = provinces
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_special_buildings.visit
			left_icon = root

			add_arcana_lifestyle_xp = travel_medium_xp
			traveler_travel_xp_effect = {
				MIN = 3
				MAX = 5
			}
			if = {
				limit = {
					is_landless_adventurer = yes
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
		}
	}
}

				has_building_with_flag = { flag = travel_point_of_interest_vounill }
			}
			add_to_list = provinces
		}
	}

	on_visit = {
		send_interface_toast = {
			title = poi_vounill.visit
			left_icon = root
			custom_tooltip = gh_unique_poi.visit.tt

			add_learning_lifestyle_xp = travel_major_xp
			add_arcana_lifestyle_xp = travel_major_xp

			if = {
				limit = {
					scope:province.barony ?= {
						is_holy_site_of = root.faith
					}

				add_piety = medium_piety_gain
			}
			else = {
				add_piety = minor_piety_gain
			}

			traveler_travel_xp_effect = {
				MIN = 3
				MAX = 5

				limit = {
					is_landless_adventurer = yes
					has_perk = organized_muster_rolls_perk
				}
				add_gold = see_how_its_done_gold_reward_value
			}
		}
	}
}

poi_events_historical = {
	build_province_list = {
		# Note, please try to keep these in chronological order
		if = { # Opakasija Landings
			limit = { game_start_date > 6265.6.1 }
			province:2394 = { add_to_list = provinces }
		}
		if = { # Death of the Preacher
			limit = { game_start_date > 6293.11.25 }
			province:1986 = { add_to_list = provinces}
		}
	}

	on_visit = {
		send_interface_toast = {
			title = travel_point_event_name_visit_message
			left_icon = root

			scope:province = {
				switch = {
					trigger = this
					# Opakasija Landings
					province:2394 = {
						add_poi_wanderer_experience_effect = yes
						stress_impact = { base = minor_stress_impact_loss }	
					}
					province:1986 = {
						add_learning_lifestyle_xp = minor_lifestyle_experience
						add_piety = major_piety_gain
					}
				}
			}

			visiting_poi_effect = yes
		}
		visiting_poi_effect = yes
	}
}
