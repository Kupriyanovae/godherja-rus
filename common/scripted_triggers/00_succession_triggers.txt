		limit = { government_allows = administrative }

			tgp_japan_single_heir_succession_override_trigger = yes
			tgp_korea_single_heir_succession_override_trigger = yes
			#has_title = title:e_hindustan
			has_title = title:h_aversarinas_aautokrata
		}
	}
	trigger_else = {
		OR = {
			AND = {
				culture = { has_cultural_pillar = heritage_marcher }
				highest_held_title_tier >= tier_kingdom
			}
			government_has_flag = government_is_landed_order
			government_has_flag = government_is_wasteland	# GH URGENT: Figure out if this is really the stopgap solution we want to use; currently just used to prevent a thousand startup errors when entering the map
			#has_title = title:e_hindustan
			has_title = title:h_aversarinas_aautokrata
			AND = {
				government_has_flag = government_is_japan_feudal
				is_independent_ruler = yes
			}
			#Add any subsequent exceptions here.
		}
	}
}
