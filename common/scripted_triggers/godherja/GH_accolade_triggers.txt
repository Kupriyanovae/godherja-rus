charioteer_attribute_trigger = {
	custom_tooltip = {
		text = charioteer_attribute.unlock_tt
		OR = {
			var:charioteer_attribute_unlock ?= yes
			has_trait_xp = {
				trait = tourney_participant
				track = horse
				value >= 30
			}
			has_trait = open_terrain_expert
		}
	}
	culture ?= { has_innovation = innovation_war_chariots }
	custom_tooltip = {
		text = single_maa_attribute_allowance
		trigger_if = {
			limit = { exists = accolade }
			accolade = { NOT = { has_accolade_category = men_at_arms } }
		}
	}
}

charioteer_attribute_acceptable_trigger = {
	OR = {
		has_trait_xp = {
			trait = tourney_participant
			track = horse
			value >= 20
		}
		has_trait = open_terrain_expert
		martial >= decent_skill_rating
	}
	culture ?= { has_innovation = innovation_war_chariots }
}

magi_attribute_trigger = {
	has_trait = magic_good
	custom_tooltip = {
		text = magi_attribute.unlock_tt
		OR = {
			var:magi_attribute_unlock ?= yes
			knows_any_magic_school_trigger = yes
			trigger_if = {
				limit = { exists = var:arcana }
				var:arcana >= skill_accolade_acceptable_value
			}
		}
	}
	custom_tooltip = {
		text = magi_attribute.unlock_tt_2
		culture = {
			OR = {
				has_innovation = innovation_early_medieval_magic
				has_innovation = innovation_imperial_magic
			}
		}
	}
	custom_tooltip = {
		text = magi_attribute.unlock_magic_illegal_tt
		NOR = {
			faith ?= { has_doctrine_parameter = magic_illegal }
			scope:owner.faith ?= { has_doctrine_parameter = magic_illegal } 
		}
	}
	custom_tooltip = {
		text = single_maa_attribute_allowance
		trigger_if = {
			limit = { exists = accolade }
			accolade = { NOT = { has_accolade_category = men_at_arms } }
		}
	}
}

magi_attribute_acceptable_trigger = {
	has_trait = magic_good
	scope:owner ?= { faith = { NOT = { has_doctrine_parameter = magic_illegal } } }
	culture ?= { has_innovation = innovation_early_medieval_magic }
}
