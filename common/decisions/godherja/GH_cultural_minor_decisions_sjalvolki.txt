### Establish Svotsadi Culture ###
establish_svotsadi_culture_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = sjalvolki

	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	desc = establish_svotsadi_culture_decision_desc
	selection_tooltip = establish_svotsadi_culture_decision_tooltip

	is_shown = {
		game_start_date <= 6298.1.1
		culture = { has_cultural_pillar = heritage_sjalvolki }
		any_sub_realm_county = {
			culture = { has_cultural_pillar = heritage_huegodoc }
		}
		NOT = {	
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_formed_svotsadi_decision
			}
		}
	}

	is_valid = {
		culture = { has_cultural_pillar = heritage_sjalvolki }
		any_sub_realm_county = {
			culture = { has_cultural_pillar = heritage_huegodoc }
		}
	}

	is_valid_showing_failures_only = {
		is_at_war = no
		is_imprisoned = no
	}

	effect = {
		hidden_effect = {
			# Generate a list of counties to flip
			every_sub_realm_county = {
				limit = {
					OR = {
						culture = { has_cultural_pillar = heritage_sjalvolki }
						culture = { has_cultural_pillar = heritage_huegodoc }
					}
				}
				add_to_list = svotsadi_counties
			}

			# Perform the culture flip
			every_in_list = {
				list = svotsadi_counties
				set_county_culture = culture:svotsadi
				holder = { set_culture = culture:svotsadi }
			}

			# Save scopes for innovation transfer
			save_scope_as = svotsadi_founder
			culture:daukeni = {save_scope_as = old_culture_1}
			culture:huegodocs = {save_scope_as = old_culture_2}
			culture:svotsadi = {
				reset_culture_creation_date = yes
				get_all_innovations_from = scope:old_culture_1
				get_all_innovations_from = scope:old_culture_2
				save_scope_as = new_culture
			}
		}

		# Player changes culture
		set_culture = culture:svotsadi
		add_character_flag = converted_culture_this_lifetime
		
		hidden_effect = {
			every_in_list = {
				list = svotsadi_counties
				limit = {
					holder.capital_county = { is_in_list = svotsadi_counties }
				}
				holder = {
					add_to_list = mass_melting_pot_list
					every_courtier = { add_to_list = mass_melting_pot_list }
					every_courtier_away = { add_to_list = mass_melting_pot_list }
				}
				every_in_list = {
					list = mass_melting_pot_list
					limit = {
						OR = {
							culture = { has_cultural_pillar = heritage_sjalvolki }
							culture = { has_cultural_pillar = heritage_huegodoc }
						}
					}
					set_culture = culture:svotsadi
				}
			}
		}

		show_as_tooltip = {
			custom_tooltip = establish_svotsadi_culture_decision_effect.tt
			culture:svotsadi = {
				get_all_innovations_from = scope:old_culture_1
				get_all_innovations_from = scope:old_culture_2
			}
		}

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_formed_svotsadi_decision
		}
	}

	cost = {}

	ai_potential = { always = yes }

	ai_will_do = {
		base = 0
		modifier = {
			add = 100
			any_sub_realm_county = {
				percent >= 0.3
				culture = culture:huegodocs
			}
		}
	}
}
