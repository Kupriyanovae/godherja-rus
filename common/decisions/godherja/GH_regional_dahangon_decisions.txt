gh_restore_bronzeport_bureaucracy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = kathuni

	ai_check_interval = 120

	desc = gh_restore_bronzeport_bureaucracy_decision_desc
	selection_tooltip = gh_restore_bronzeport_bureaucracy_decision_tooltip

	is_shown = {
		#Standard filter checks.
		is_landed = yes
		exists = dynasty
		NOT = {
			government_has_flag = government_is_republic_playable
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:gh_restore_bronzeport_bureaucracy_decision
			}
		}
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_continental_clan
		}
		OR = {
			has_primary_title = title:k_bronzeport
			any_liege_or_above = {
				has_title = title:k_bronzeport
			}
		}
	}

	is_valid_showing_failures_only = {
		stewardship >= 25
		learning >= 15
		prestige_level >= 4
		is_at_war = no
		NOT = { exists = involved_activity }
		is_imprisoned = no
	}

	is_valid = {
		#Must be independent.
		is_independent_ruler = yes
		#Must have the Bronzeport
		has_primary_title = title:k_bronzeport
		#Must have the highest level crown authority.
		has_realm_law = crown_authority_3
	}

	effect = {
		save_scope_as = bronzeport_bureau_restorer
		gh_restore_bronzeport_bureaucracy_decision_effect = yes
		initialize_default_cliques_effect = yes
		trigger_event = dahangon_decision.0001
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:gh_restore_bronzeport_bureaucracy_decision
		}
		set_global_variable = {
			name = gh_restore_bronzeport_bureaucracy_decision
			value = scope:bronzeport_bureau_restorer
		}
	}

	cost = {
		gold = 600
		prestige = 1000
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 20
	}
}

building_settlements_in_rovarska_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_ck2_china_stable.dds"
	}
	decision_group_type = kathuni

	cooldown = {
		years = 6
	}
	title = building_settlements_in_rovarska_decision.t
	desc = building_settlements_in_rovarska_decision_desc
	ai_check_interval = 40
	is_shown = {
		OR = {
			has_title = title:e_yuankang
			has_title = title:k_bronzeport
		}
		culture = {
			has_cultural_pillar = heritage_easterner
		}
	}

	cost = {
		gold = 400
	}

	is_valid = {
		is_at_war = no
		scope:barony.county = {
			county_control >= 80
		}
		scope:barony.title_province  = {
			geographical_region = world_ga_rovarska_archipelago_formable
		}
		
		scope:barony.county.holder = {
			OR = {
				this = root
				opinion = {
					target = root
					value >= 60
				}
				AND = {
					culture = {
						has_cultural_pillar = heritage_easterner
					}
					opinion = {
						target = root
						value >= 25
					}
				}
			}
		}
		OR = {
			has_title = title:e_yuankang
			has_title = title:k_bronzeport
		}
	}

	widget = {
		gui = decision_view_widget_gh_bring_farmers_to_county
		controller = create_holy_order
		barony_valid = {
			trigger_if = {
				limit = { exists = this }
				barony = {
					is_capital_barony = yes
				}
				county = {
					NOT = {
						culture = {
							has_cultural_pillar = heritage_easterner
						}
					}
					any_county_province = {
						geographical_region = world_ga_rovarska_archipelago_formable
					}
				}
			}
			trigger_else = {
				custom_description = {
					text = "has_gh_barony_county_selected"
					always = no
				}
			}
		}
	}

	effect = {
		scope:barony.county = {
			set_county_culture = culture:tekauni
			set_county_faith = root.faith
			change_development_level = 3
			add_county_modifier = {
				modifier = rovarska_trade_stronghold_modifier
				years = 15
			}
			this.title_province = {
				gh_change_farmland_building_effect = yes
			}
		}
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = unavailable_unique_decisions
						target = flag:building_settlements_in_rovarska_decision
					}
				}
			}
			trigger_event = dahangon_decision.0015
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:building_settlements_in_rovarska_decision
			}	
		}
	}

	ai_will_do = {
		base = 15
	}
}

			}

					government_has_flag = government_is_tribal

			}
		}

			add_building = hill_forts_01
		}
