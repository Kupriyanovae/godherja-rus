create_glass_artifact_decision = { # Create a Glass Artifact
	picture = { reference = "gfx/interface/illustrations/decisions/decision_smith.dds" }
	cooldown = { years = 5 }
	sort_order = 0
	
	is_shown = {
		is_human = yes
		culture = { has_cultural_parameter = can_make_glass_artifacts }
	}
	
	is_valid_showing_failures_only = {
		is_at_war = no
		available_for_personal_inspiration_trigger = yes
		custom_tooltip = {
			text = already_has_a_personal_inspiration
			NOT = { has_variable = personal_inspiration_story }
		}
	}

	minimum_cost = {
		gold = {
			value = basic_fund_inspiration_cost
			if = {
				limit = { domicile ?= { has_domicile_parameter = estate_reduce_commission_artifact_cost } }
				multiply = estate_reduce_commission_artifact_cost_value
			}
			if = {
				limit = { confederation ?= { has_cohesion_level_parameter = any_member_commission_and_inspiration_reduced } }
				multiply = {
					value = 0.9
					desc = CB_COHESION_PARAMETER_DISCOUNT
				}
			}
			if = {
				limit = { has_character_modifier = mpo_artifact_material_modifier }
				multiply = 0
			}
			if = { ## GH ADDITION: Cognition Perk discount bonus
				limit = { has_perk = arcana_cognition_perk4_tree2 }
				multiply = 0.66
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = create_glass_artifact_decision_effect
			set_variable = {
				name = personal_inspiration_type
				value = flag:glassware_inspiration
			}
			set_variable = {
				name = personal_inspiration_initial_gold_investment
				value = {
					value = basic_fund_inspiration_cost
					if = {
						limit = { domicile ?= { has_domicile_parameter = estate_reduce_commission_artifact_cost } }
						multiply = estate_reduce_commission_artifact_cost_value
					}
					if = {
						limit = { confederation ?= { has_cohesion_level_parameter = any_member_commission_and_inspiration_reduced } }
						multiply = {
							value = 0.9
							desc = CB_COHESION_PARAMETER_DISCOUNT
						}
					}
					if = {
						limit = { has_character_modifier = mpo_artifact_material_modifier }
						multiply = 0
					}
					if = { ## GH ADDITION: Cognition Perk discount bonus
						limit = { has_perk = arcana_cognition_perk4_tree2 }
						multiply = 0.66
					}
				}
			}
			create_story = story_cycle_personal_inspiration
		}
		custom_description_no_bullet = { text = create_glass_artifact_decision_inspiration_tt }
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 36
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}
	
	ai_potential = {
		ai_greed < medium_positive_ai_value
		short_term_gold > ai_war_chest_desired_gold_value
		war_chest_gold >= halved_ai_war_chest_gold_maximum
		NOT = { has_variable = conqueror }
		ai_should_focus_on_building_in_their_capital = no
	}

	ai_will_do = {
		base = 50
	}
}
