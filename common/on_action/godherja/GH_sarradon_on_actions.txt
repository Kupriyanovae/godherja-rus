gh_sarradonian_desert_yearly_pulse = {
	effect = {
		# Sand-Devil (The Howling) - Catastrophic storm season
		if = {
			limit = {
				situation_top_has_catalyst = catalyst_situation_passing_of_time_sand_devil
			}
			random = {
				chance = 3 # Base chance: 3% per year
				# Increase slightly with game time, but never guarantee
				modifier = {
					years_from_game_start >= 50
					add = 1
				}
				modifier = {
					years_from_game_start >= 100
					add = 1
				}
				modifier = {
					years_from_game_start >= 200
					add = 1
				}
				modifier = {
					years_from_game_start >= 300
					add = 1
				}
				modifier = {
					years_from_game_start >= 500
					add = 1
				}
				modifier = {
					years_from_game_start >= 750
					add = 1
				}
				# Conditions that make Sand-Devil more likely
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_scorching_zenith_season
					}
					add = 2
				}
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_scorching_zenith_season
					}
					add = 3
				}
				trigger_situation_catalyst = catalyst_situation_passing_of_time_sand_devil
			}
		}
		
		# Celestial Stillness (Oasis of Stars) - Rare beneficial season
		if = {
			limit = {
				situation_top_has_catalyst = catalyst_situation_passing_of_time_celestial_stillness
			}
			random = {
				chance = 2 # Base chance: 2% per year (even rarer than Sand-Devil)
				# Increase slightly with game time
				modifier = {
					years_from_game_start >= 100
					add = 1
				}
				modifier = {
					years_from_game_start >= 250
					add = 1
				}
				modifier = {
					years_from_game_start >= 500
					add = 1
				}
				modifier = {
					years_from_game_start >= 750
					add = 1
				}
				# Conditions that make Celestial Stillness more likely
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_veil_thin_nights_season
					}
					add = 1
				}
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_sand_scribe_season
					}
					add = 1
				}
				modifier = {
					any_situation_sub_region = {
						OR = {
							sub_region_current_phase = situation_sand_veil_thin_nights_season
							sub_region_current_phase = situation_sand_sand_scribe_season
						}
					}
					add = 2
				}
				# Make extremely unlikely if recently had Sand-Devil
				modifier = {
					has_global_variable = {
						name = had_sand_devil_var
					}
					factor = 0.1 # 90% reduction
				}
				# Make subsequent occurrences slightly less likely (opposite of Sand-Devil)
				modifier = {
					has_global_variable = had_celestial_stillness_var
					factor = 0.5 # Each previous occurrence halves the chance
				}
				trigger_situation_catalyst = catalyst_situation_passing_of_time_celestial_stillness
			}
		}
		
		# Lucky Chance Sand-Devil - Random catastrophic events
		if = {
			limit = {
				situation_top_has_catalyst = catalyst_situation_passing_of_time_lucky_chance_sand_devil
			}
			random = {
				chance = 1 # Base chance: 1% per year for "lucky" (unlucky) trigger
				# Rare conditions that might trigger unexpected Sand-Devil
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_scorching_zenith_season
					}
					add = 2
				}
				# Full Blood Moon
				modifier = {
					global_var:current_moon_phase.var:moon_phase_name ?= flag:full_moon_blood_moon
					add = 30
				}
				trigger_situation_catalyst = catalyst_situation_passing_of_time_lucky_chance_sand_devil
			}
		}
		
		# Lucky Chance Celestial Stillness - Random beneficial events
		if = {
			limit = {
				situation_top_has_catalyst = catalyst_situation_passing_of_time_lucky_chance_celestial_stillness
			}
			random = {
				chance = 0.5 # Base chance: 0.5% per year for "lucky" trigger
				# Conditions for unexpected Celestial Stillness
				modifier = {
					any_situation_sub_region = {
						sub_region_current_phase = situation_sand_sand_devil_season
					}
					add = 2 # The storm suddenly clearing into perfect stillness
				}
				# Full Moon
				modifier = {
					global_var:current_moon_phase.var:moon_phase_name ?= flag:full_moon
					add = 30
				}
				trigger_situation_catalyst = catalyst_situation_passing_of_time_lucky_chance_celestial_stillness
			}
		}
	}
}

		}

gh_sarradonian_desert_yearly_pulse_random = {
	effect = {
		if = {
			limit = {
				is_ai = yes
				is_at_war = no
				government_has_flag = government_is_nomadic
			}
			save_temporary_scope_as = nomad_capital_ruler
			if = { # Count-tier nomads over their domain limit should abandon low-fertility counties to herders
				limit = {
					primary_title.tier = tier_county
					domain_size > domain_limit
					this = top_liege
					any_held_title = {
						tier = tier_county
						is_landless_type_title = no
						this != scope:nomad_capital_ruler.capital_county
						OR = {
							any_neighboring_county = {
								holder.top_liege != scope:nomad_capital_ruler.top_liege
							}
							is_coastal_county = yes
						}
						NOT = {
							any_neighboring_county = {
								this = scope:nomad_capital_ruler.capital_county
							}
						}
					}
				}
				alternative_limit = {
					primary_title.tier = tier_county
					domain_size > domain_limit
					this = top_liege
					any_held_title = {
						tier = tier_county
						is_landless_type_title = no
						this != scope:nomad_capital_ruler.capital_county
						OR = {
							any_neighboring_county = {
								holder.top_liege != scope:nomad_capital_ruler.top_liege
							}
							is_coastal_county = yes
						}
					}
				}
				random_held_title = {
					limit = {
						tier = tier_county
						is_landless_type_title = no
						this != scope:nomad_capital_ruler.capital_county
						OR = {
							any_neighboring_county = {
								holder.top_liege != scope:nomad_capital_ruler.top_liege
							}
							is_coastal_county = yes
						}
					}
					weight = {
						base = 100
						modifier = { #Less likely to pick someone you actively dislike
							add = {
								value = county_fertility
								multiply = -1
							}
							uses_county_fertility = yes
						}
					}
					save_scope_as = title_to_abandon
				}
				if = { # In the Desert, if the county has terrible fertility then it turns to wasteland rather than herders
					limit = {
						scope:title_to_abandon.county_fertility <= terrible_county_fertility_level
					}
					spawn_ruin = { COUNTY = scope:title_to_abandon }
				}
				else = {
					create_character = {
						template = herder_character
						dynasty = none
						location = scope:nomad_capital_ruler.location
						save_scope_as = new_herder
					}
					scope:title_to_abandon = {
						if = {
							limit = {
								title_province = {
									has_holding_type = nomad_holding
								}
							}
							title_province = {
								set_holding_type = herder_holding
							}
						}
						create_title_and_vassal_change = {
							type = granted
							save_scope_as = change
							add_claim_on_loss = no
						}
						change_title_holder_include_vassals = {
							holder = scope:new_herder
							change = scope:change
						}
						resolve_title_and_vassal_change = scope:change
					}
					scope:new_herder = {
						change_government = herder_government
						every_held_title = {
							limit = {
								is_landless_type_title = yes
							}
							scope:new_herder = { destroy_title = prev }
						}
					}
					start_tributary_interaction_effect = {
						TRIBUTARY = scope:new_herder
						SUZERAIN = scope:nomad_capital_ruler
					}
					scope:nomad_capital_ruler = {
						add_prestige = minor_prestige_loss
					}
				}
			}
			random = {
				chance = {
					value = 5
					if = {
						limit = {
							prestige <= 600
						}
						add = 30
					}
					if = {
						limit = {
							prestige <= 100
						}
						add = 70
					}
				}
				ai_attempt_to_host_activity = activity_hunt
			}
			random = {
				chance = {
					value = 0
					if = {
						limit = {
							has_character_modifier = mpo_si_slaughtered_animals
						}
						add = 30
					}
					if = {
						limit = {
							gold > prestige
							gold > 200
							prestige < 600
						}
						add = 30
					}
				}
				ai_attempt_to_host_activity = activity_feast
			}
			# Overrun Kingdoms but for Desert Nomads
			if = {
				limit = {
					# No Havsaran zud, and less likely to flee when hard times happen
					is_landed = no
					domicile.herd >= 10000
				}
				random = {
					chance = 50
					ai_start_best_war = {
						cb = {
							mpo_nomad_invasion_cb
						}
						is_valid = {
							scope:target_character = {
								current_military_strength < scope:nomad_capital_ruler.current_military_strength
								OR = {
									gold < scope:nomad_capital_ruler.gold
									current_military_strength <= scope:nomad_capital_ruler.fifty_percent_of_current_military_strength
								}
							}
						}
					}
				}
			}
			# Big Desert Nomads should try to expand
			if = {
				limit = {
					highest_held_title_tier >= tier_kingdom
					war_chest_gold >= war_chest_gold_maximum
				}
				random = {
					chance = 50
					ai_start_best_war = {
						cb = {
							de_jure_cb_nomadic
							retaliation_cb
							humiliation_cb
							make_tributary_cb
							sovereignty_cb
							nomadic_conquest_duchy_cb
						}
						is_valid = {
							scope:target_character = {
								current_military_strength < scope:nomad_capital_ruler.current_military_strength
								OR = {
									gold < scope:nomad_capital_ruler.gold
									current_military_strength <= scope:nomad_capital_ruler.fifty_percent_of_current_military_strength
								}
							}
						}
					}
				}
			}
			# Clean-up Landless AI Nomads who have lost
			clean_up_ai_nomad_effect = yes
		}
	}
}
